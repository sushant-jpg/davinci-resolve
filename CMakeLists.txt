cmake_minimum_required(VERSION 3.20)
project(SentinelStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Multimedia MultimediaWidgets Concurrent)
add_library(sentinel_engine
  src/core/Project.cpp src/core/Editor.cpp src/project/ProjectIO.cpp
  src/media/MediaProbe.cpp src/export/Exporter.cpp)
target_include_directories(sentinel_engine PUBLIC src)
target_link_libraries(sentinel_engine PUBLIC Qt6::Core)
add_library(sentinel_desktop src/ui/TimelineWidget.cpp src/ui/MainWindow.cpp src/ui/MainWindowProject.cpp src/ui/MainWindowMedia.cpp src/playback/PlaybackController.cpp)
target_link_libraries(sentinel_desktop PUBLIC sentinel_engine Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Concurrent)
add_executable(sentinel-studio src/app/main.cpp)
target_link_libraries(sentinel-studio PRIVATE sentinel_desktop)
foreach(target sentinel_engine sentinel_desktop sentinel-studio)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(sentinel_tests tests/EngineTests.cpp)
  target_link_libraries(sentinel_tests PRIVATE sentinel_engine Qt6::Test)
  add_test(NAME engine COMMAND sentinel_tests)
  add_executable(sentinel_ui_tests tests/UiTests.cpp)
  target_link_libraries(sentinel_ui_tests PRIVATE sentinel_desktop Qt6::Test)
  add_test(NAME desktop COMMAND sentinel_ui_tests)
  set_tests_properties(engine desktop PROPERTIES TIMEOUT 120)
  set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic;QT_STYLE_OVERRIDE=Fusion")
endif()
install(TARGETS sentinel-studio RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES assets/sentinel-studio.desktop DESTINATION share/applications)
  install(FILES assets/sentinel-studio.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
